// Matmul maps: C[m][n] += A[m][k] * B[k][n]
// Loop order: d0=m (parallel), d1=k (reduction), d2=n (parallel).
#map_mm_a = affine_map<(d0, d1, d2) -> (d0, d1)>
#map_mm_b = affine_map<(d0, d1, d2) -> (d1, d2)>
#map_mm_c = affine_map<(d0, d1, d2) -> (d0, d2)>

module {
  func.func private @printMemrefF32(memref<*xf32>)

  // C[2x2] = A[2x3] × B[3x2]
  //
  // A = [[1, 2, 3],     B = [[7,  8],      C = [[58,  64],
  //      [4, 5, 6]]          [9,  10],           [139, 154]]
  //                          [11, 12]]
  func.func @matmul(%A: memref<2x3xf32>,
                    %B: memref<3x2xf32>,
                    %C: memref<2x2xf32>) {
    linalg.generic {
      indexing_maps = [#map_mm_a, #map_mm_b, #map_mm_c],
      iterator_types = ["parallel", "reduction", "parallel"]
    } ins(%A, %B : memref<2x3xf32>, memref<3x2xf32>)
      outs(%C : memref<2x2xf32>) {
    ^bb0(%a: f32, %b: f32, %c: f32):
      %mul = arith.mulf %a, %b : f32
      %add = arith.addf %mul, %c : f32
      linalg.yield %add : f32
    }
    return
  }

  func.func @main() {
    %A = memref.alloc() : memref<2x3xf32>
    %B = memref.alloc() : memref<3x2xf32>
    %C = memref.alloc() : memref<2x2xf32>

    // A = [[1, 2, 3], [4, 5, 6]]
    %f1  = arith.constant  1.0 : f32
    %f2  = arith.constant  2.0 : f32
    %f3  = arith.constant  3.0 : f32
    %f4  = arith.constant  4.0 : f32
    %f5  = arith.constant  5.0 : f32
    %f6  = arith.constant  6.0 : f32
    affine.store %f1,  %A[0, 0] : memref<2x3xf32>
    affine.store %f2,  %A[0, 1] : memref<2x3xf32>
    affine.store %f3,  %A[0, 2] : memref<2x3xf32>
    affine.store %f4,  %A[1, 0] : memref<2x3xf32>
    affine.store %f5,  %A[1, 1] : memref<2x3xf32>
    affine.store %f6,  %A[1, 2] : memref<2x3xf32>

    // B = [[7, 8], [9, 10], [11, 12]]
    %f7  = arith.constant  7.0 : f32
    %f8  = arith.constant  8.0 : f32
    %f9  = arith.constant  9.0 : f32
    %f10 = arith.constant 10.0 : f32
    %f11 = arith.constant 11.0 : f32
    %f12 = arith.constant 12.0 : f32
    affine.store %f7,  %B[0, 0] : memref<3x2xf32>
    affine.store %f8,  %B[0, 1] : memref<3x2xf32>
    affine.store %f9,  %B[1, 0] : memref<3x2xf32>
    affine.store %f10, %B[1, 1] : memref<3x2xf32>
    affine.store %f11, %B[2, 0] : memref<3x2xf32>
    affine.store %f12, %B[2, 1] : memref<3x2xf32>

    // C starts as zeros.
    %f0 = arith.constant 0.0 : f32
    affine.store %f0, %C[0, 0] : memref<2x2xf32>
    affine.store %f0, %C[0, 1] : memref<2x2xf32>
    affine.store %f0, %C[1, 0] : memref<2x2xf32>
    affine.store %f0, %C[1, 1] : memref<2x2xf32>

    call @matmul(%A, %B, %C) : (memref<2x3xf32>, memref<3x2xf32>, memref<2x2xf32>) -> ()

    %C_u = memref.cast %C : memref<2x2xf32> to memref<*xf32>
    call @printMemrefF32(%C_u) : (memref<*xf32>) -> ()

    memref.dealloc %A : memref<2x3xf32>
    memref.dealloc %B : memref<3x2xf32>
    memref.dealloc %C : memref<2x2xf32>
    return
  }
}
